# Runner used for each run (each evidence file spells out the equivalent commands):
#!/bin/bash
# run.sh <name> <mode> <prompt> [extra agy args...]
S=$SCRATCH
H=${H:-$S/agy/home-deny}
name=$1; mode=$2; prompt=$3; shift 3
R=$S/agy/deny-runs/$name; rm -rf $R; mkdir -p $R/ws
printf '%s' "$mode" > $H/PROBE_MODE
rm -rf $H/probe-log
printf "alpha beta gamma\n" > $R/ws/edit-me.txt
cd $R/ws
HOME=$H AGY_CLI_DISABLE_AUTO_UPDATE=1 timeout 300 agy -p "$prompt" --model gemini-3.8-flash-low --output-format stream-json "$@" > $R/stream.jsonl 2> $R/stderr.txt
echo "[exit $?]" > $R/exit.txt
mv $H/probe-log $R/probe-log 2>/dev/null
ls -la $R/ws > $R/ws-listing.txt; cat $R/ws/edit-me.txt >> $R/ws-listing.txt

# Runs were made one at a time: PROBE_MODE and probe-log live in the shared home, so parallel runs would mix logs.
