#!/usr/bin/env node
// Fail-mode PreToolUse variant: $HOME/FAIL_MODE = empty | emptyobj | exit1. Logs stdin to $HOME/probe-log.
import fs from 'node:fs'; import path from 'node:path'; import os from 'node:os';
const home = os.homedir(); let mode = 'empty';
try { mode = fs.readFileSync(path.join(home, 'FAIL_MODE'), 'utf8').trim(); } catch {}
let raw = ''; process.stdin.setEncoding('utf8');
process.stdin.on('data', (c) => { raw += c; });
process.stdin.on('end', () => {
  const dir = path.join(home, 'probe-log'); fs.mkdirSync(dir, { recursive: true });
  fs.writeFileSync(path.join(dir, `${Date.now()}-${process.pid}-PreToolUse-fail-${mode}.json`), raw);
  if (mode === 'emptyobj') process.stdout.write('{}\n');
  else if (mode === 'exit1') { process.stderr.write('probe fail boom\n'); process.exit(1); }
  // 'empty': print nothing, exit 0
});
