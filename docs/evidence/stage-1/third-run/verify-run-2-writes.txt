// cartkit: a tiny shopping cart. Prices are integer cents.
export function createCart(items = []) {
  return { items: items.map(i => ({ ...i })) }
}

export function addItem(cart, item) {
  cart.items.push({ ...item })
  return cart
}

export function subtotal(cart) {
  let sum = 0
  for (const item of cart.items) sum += item.price * item.qty
  return sum
}

export function findItem(cart, id) {
  return cart.items.find(i => i.id === id)
}

export function removeItem(cart, id) {
  const idx = cart.items.findIndex(i => i.id === id)
  if (idx !== -1) cart.items.splice(idx, 1)
  return cart
}

export function parseQty(text) {
  return parseInt(text, 10)
}

// Takes `percent` (clamped to 0..100) off `totalCents`; returns the discount in integer cents.
export function applyDiscount(totalCents, percent) {
  const p = Number.isFinite(percent) ? Math.min(100, Math.max(0, percent)) : 0
  return Math.round(totalCents * p / 100)
}

const COUPONS = { SAVE10: { percent: 10, expires: '2026-01-01' }, HALF: { percent: 50, expires: '2027-06-30' } }

export function isExpired(coupon, now = Date.now()) {
  return new Date(coupon.expires).getTime() <= now
}

export function couponPercent(code, now = Date.now()) {
  if (typeof code !== 'string' || !Object.hasOwn(COUPONS, code)) return 0
  const coupon = COUPONS[code]
  return isExpired(coupon, now) ? 0 : coupon.percent
}

export function taxOn(cents, rate) {
  if (!Number.isFinite(rate) || rate < 0) throw new RangeError('taxRate must be a non-negative number')
  return Math.round(cents * rate)
}

export function addTax(subtotalCents, rate) {
  return subtotalCents + taxOn(subtotalCents, rate)
}

export function formatPrice(cents) {
  return '$' + (cents / 100).toFixed(2)
}

export async function saveCart(cart, store) {
  await store.put('cart', JSON.stringify({ cart }))
}

export function checkout(cart, { coupon, taxRate = 0, now = Date.now() } = {}) {
  const sub = subtotal(cart)
  const discount = applyDiscount(sub, couponPercent(coupon, now))
  const tax = taxOn(sub - discount, taxRate)
  return { subtotal: sub, discount, tax, total: sub - discount + tax }
}
