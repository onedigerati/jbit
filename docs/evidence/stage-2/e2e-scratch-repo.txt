/**
 * Turn text into a URL slug: lowercase ASCII letters and digits joined by single hyphens.
 * Accents are stripped first; anything else (punctuation, spaces, hyphens, underscores) acts as a separator.
 */
export function slugify(text) {
  return String(text)
    .normalize("NFKD")
    .replace(/[\u0300-\u036f]/g, "") // strip accents: "é" -> "e"
    .toLowerCase()
    .replace(/['’]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");
}
import { test } from "node:test";
import assert from "node:assert/strict";
import { slugify } from "./slugify.mjs";

test("lowercases, drops punctuation, joins words with single hyphens", () => {
  assert.equal(slugify("Hello, World!"), "hello-world");
});

test("collapses leading/trailing spaces and repeated separators", () => {
  assert.equal(slugify("  a -- b  "), "a-b");
});

test("empty string gives empty slug", () => {
  assert.equal(slugify(""), "");
});

test("only punctuation gives empty slug", () => {
  assert.equal(slugify("!!! --- ???"), "");
});

test("keeps digits and drops apostrophes inside words", () => {
  assert.equal(slugify("Top 10 Tips: Don't Panic"), "top-10-tips-dont-panic");
});

test("strips accents rather than dropping the letter", () => {
  assert.equal(slugify("Café Crème"), "cafe-creme");
});
