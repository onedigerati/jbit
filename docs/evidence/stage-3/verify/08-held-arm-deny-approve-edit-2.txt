const ALLOWED_USERS: Record<string, string> = {
  admin: "admin123",
  viewer: "viewer123",
};

export function login(user: string, pass: string): boolean {
  if (!user || typeof user !== "string" || user.trim().length === 0) {
    return false;
  }
  if (!pass || typeof pass !== "string" || pass.length === 0) {
    return false;
  }
  const cleanUser = user.trim();
  if (Object.prototype.hasOwnProperty.call(ALLOWED_USERS, cleanUser)) {
    return ALLOWED_USERS[cleanUser] === pass;
  }
  return false;
}
