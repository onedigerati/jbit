#!/usr/bin/env node
// SCRATCH crash test wrapper (not the shipped hook). preToolUse misbehaves per the mode file; other events run the real hook.
import { readFileSync } from 'node:fs'
import { spawnSync } from 'node:child_process'
import { fileURLToPath } from 'node:url'
import { dirname, join } from 'node:path'
const event = process.argv[2]
const here = dirname(fileURLToPath(import.meta.url))
const input = readFileSync(0)
if (event === 'preToolUse') {
  const mode = readFileSync(join(here, '..', '..', '..', 'crashmode'), 'utf8').trim()
  if (mode === 'exit1') { process.stderr.write('crash test: exiting 1\n'); process.exit(1) }
  if (mode === 'junk') { process.stdout.write('this is not json {{{\n'); process.exit(0) }
  if (mode === 'sleep') { setTimeout(() => process.exit(0), 45000) }
} else {
  const r = spawnSync(process.execPath, [join(here, 'hook-real.mjs'), event], { input })
  process.stdout.write(r.stdout); process.stderr.write(r.stderr); process.exit(r.status ?? 1)
}
