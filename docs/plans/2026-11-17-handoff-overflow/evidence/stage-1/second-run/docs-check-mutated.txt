--- mutate.sh
#!/bin/bash
# For each role agent: back it up, cut its handoff rule, run the docs check, restore it, run the docs check again.
cd ~/src/repos/jbit
declare -A CUT=(
  [builder]=" When the detail doesn't fit in 20 lines[^.]*\.[^.]*left out\."
  [researcher]=" When the detail doesn't fit in 20 lines[^.]*left out\."
  [verifier]=" Then, under a \`## Report\` heading[^;]*;[^.]*\`report\.md\`\."
  [reviewer]=" Write the full review to[^.]*\.[^.]*\.[^:]*:[^.]*\."
)
for role in builder verifier reviewer researcher; do
  f=plugins/jbit/agents/$role.md
  cp $f /tmp/$role.md.bak
  echo "## Mutated: handoff rule cut from $role.md"
  sed -i -E "s/${CUT[$role]}//" $f
  echo "lines changed vs original: $(diff /tmp/$role.md.bak $f | grep -c '^>')"
  node --test plugins/jbit/tests/docs.check.mjs 2>&1 | grep -E "^not ok|^  error|^# (pass|fail)"
  cp /tmp/$role.md.bak $f
  echo "## Restored $role.md"
  node --test plugins/jbit/tests/docs.check.mjs 2>&1 | grep -E "^not ok|^# (pass|fail)"
  echo
done
