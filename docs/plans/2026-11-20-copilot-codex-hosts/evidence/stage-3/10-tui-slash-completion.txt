# tui-complete.py (scratchpad):
# Opens Copilot's TUI in a pty, accepts the folder-trust prompt, types "/jbit:" WITHOUT Enter to show the slash-command
# completion menu, captures the raw screen bytes, then quits with Ctrl-C. No prompt is submitted.
import os, pty, sys, time, select, re
home, proj, out = sys.argv[1], sys.argv[2], sys.argv[3]  # argv[4]: the text to type
env = dict(os.environ, COPILOT_HOME=home, TERM='xterm-256color', COLUMNS='160', LINES='50')
pid, fd = pty.fork()
if pid == 0:
    os.chdir(proj)
    os.execvpe('copilot', ['copilot'], env)
import fcntl, termios, struct
fcntl.ioctl(fd, termios.TIOCSWINSZ, struct.pack('HHHH', 50, 160, 0, 0))
buf = b''
def pump(secs):
    global buf
    end = time.time() + secs
    while time.time() < end:
        r, _, _ = select.select([fd], [], [], 0.2)
        if r:
            try: buf += os.read(fd, 65536)
            except OSError: return
pump(8)
marks = {}
marks['before_trust'] = len(buf)
if b'trust' in buf.lower():
    os.write(fd, b'\r'); pump(5)
marks['before_type'] = len(buf)
q = sys.argv[4].encode()
os.write(fd, q); pump(4)  # the whole query at once, never Enter
os.write(fd, b'\x15'); pump(1)  # clear the line
marks['after_type'] = len(buf)
os.write(fd, b'\x03'); pump(1); os.write(fd, b'\x03'); pump(2)
try: os.kill(pid, 9)
except Exception: pass
open(out, 'wb').write(buf)
txt = re.sub(rb'\x1b\[[0-9;?<>=$]*[ -/]*[@-~]|\x1b\][^\x07\x1b]*(\x07|\x1b\\)|\x1b[=>PX^_].', b' ', buf[marks['before_type']:marks['after_type']])
print(re.sub(rb'[ \t]+', b' ', txt).decode('utf8', 'replace'))
