# Mutation runs of the image gate on scratch copies of plugins/jbit. Never touches the repo.
S=<scratchpad>/mut
SRC=~/src/repos/jbit/plugins/jbit
mkdir -p $S
mut() {
  name=$1; expr=$2; d=$S/$name
  rm -rf $d; mkdir -p $d
  cp -r $SRC/. $d/
  perl -0pi -e "$expr" $d/hooks/register.ts
  if cmp -s $SRC/hooks/register.ts $d/hooks/register.ts; then echo "== $name: MUTATION NOT APPLIED"; return; fi
  echo "--- $name diff:"; diff $SRC/hooks/register.ts $d/hooks/register.ts | head -8
  (cd $d && claude plugin test . 2>&1) > $S/$name.out
  echo "== $name: $(grep -E '^ *[0-9]+ (pass|fail)$' $S/$name.out | tr '\n' ' ')"
  grep -E '^\(fail\)' $S/$name.out | head -20
}
mut off 's/(export async function imageDenial\([^\n]*\{\n)/$1  return undefined\n/'
